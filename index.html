@Insert(onConflict = OnConflictStrategy.REPLACE) suspend fun insertOrUpdateProgress(progress: UserProgressEntity) @Query("SELECT * FROM lesson_records") fun getAllLessonRecordsFlow(): Flow> @Query("SELECT * FROM lesson_records WHERE lessonId = :lessonId") suspend fun getLessonRecord(lessonId: String): LessonRecordEntity?
